cmake_minimum_required(VERSION 3.25)
project(ck_fs_installed_layer_consumers LANGUAGES CXX)
option(CK_CORE_FS_CONSUMER_MATCH_TEST_PRODUCER
  "Private qualification only: match a BUILD_TESTING=ON producer" OFF)
find_package(ck-core-fs CONFIG REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IS_ABSOLUTE "${CK_LAYER_FIXTURE_ROOT}")
  message(FATAL_ERROR "CK_LAYER_FIXTURE_ROOT must be an explicit absolute temporary directory")
endif()
enable_testing()
add_executable(core_consumer core.cpp)
target_link_libraries(core_consumer PRIVATE ck::ck_core_fs_engine)
add_test(NAME installed_core COMMAND core_consumer)
add_executable(platform_consumer native.cpp)
target_compile_definitions(platform_consumer PRIVATE CK_LAYER_PLATFORM_ONLY=1)
target_link_libraries(platform_consumer PRIVATE ck::ck_files_platform)
add_executable(native_consumer native.cpp)
target_link_libraries(native_consumer PRIVATE ck::ck_core_fs_native)
add_executable(capi_consumer capi.cpp)
target_link_libraries(capi_consumer PRIVATE ck::ck_file_engine_c)
foreach(layer core platform native capi)
  if(CK_CORE_FS_CONSUMER_MATCH_TEST_PRODUCER)
    target_compile_definitions(${layer}_consumer PRIVATE
      CK_FS_INTERNAL_TEST_HOOKS=1 CK_FS_LEGACY_NATIVE_TEST_OBSERVERS)
    if(WIN32)
      target_compile_definitions(${layer}_consumer PRIVATE CK_FS_NATIVE_CLOSE_TEST_HOOKS)
    endif()
  endif()
  if(MSVC)
    target_compile_options(${layer}_consumer PRIVATE /UNDEBUG)
  else()
    target_compile_options(${layer}_consumer PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_link_options(${layer}_consumer PRIVATE "LINKER:-Map=${CMAKE_CURRENT_BINARY_DIR}/${layer}.map")
    endif()
  endif()
endforeach()
foreach(layer platform native capi)
  add_test(NAME installed_${layer} COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/run_fixture.py $<TARGET_FILE:${layer}_consumer> ${CK_LAYER_FIXTURE_ROOT})
endforeach()
set_tests_properties(installed_core installed_platform installed_native installed_capi PROPERTIES TIMEOUT 90)
